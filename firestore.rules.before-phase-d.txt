rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId} {

      // User hanya boleh membaca profilnya sendiri
      allow read: if request.auth != null
                  && request.auth.uid == userId;

      // User hanya boleh membuat dokumen profil miliknya sendiri
      // dan tidak boleh menentukan role selain "user"
      allow create: if request.auth != null
                    && request.auth.uid == userId
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.role == "user";

      // User boleh update profil sendiri,
      // tapi UID dan role tidak boleh diubah dari client
      allow update: if request.auth != null
                    && request.auth.uid == userId
                    && request.resource.data.uid == resource.data.uid
                    && request.resource.data.role == resource.data.role;

      // Jangan izinkan client menghapus user profile dulu
      allow delete: if false;
    }
  }
}
